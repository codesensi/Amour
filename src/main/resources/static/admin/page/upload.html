使用 layui 自带 upload 组件，上传地址经统一请求层管理。
当前为 mock 模式：静态资源容器不支持 POST 上传，请将 js/api-config.js 中 mock 设为 false 后对接真实后端体验完整上传。
选择文件
        // 组件经统一请求层接入：真实模式请求 /upload，自动携带 token
        upload.render({
            elem: '#uploadBtn',
            url: layui.api.url('upload'),
            field: 'file',               // 与后端 multipart 字段名一致
            accept: 'file',
            multiple: true,
            drag: true,
            done: function (res) { /* 统一响应 Result 由 parseData 解析 */ },
            error: function () { layer.msg('上传失败'); }
        });
